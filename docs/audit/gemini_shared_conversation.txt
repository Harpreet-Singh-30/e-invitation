src/components/Navbar.tsx

TypeScript
"use client";
import { useState, useEffect } from "react";
import { motion, AnimatePresence } from "framer-motion";
import { weddingData } from "@/data/wedding";
import { Menu, X } from "lucide-react";

export default function Navbar() {
  const [scrolled, setScrolled] = useState(false);
  const [isOpen, setIsOpen] = useState(false);

  useEffect(() => {
    const handleScroll = () => {
      setScrolled(window.scrollY > 50);
    };
    window.addEventListener("scroll", handleScroll);
    return () => window.removeEventListener("scroll", handleScroll);
  }, []);

  const links = [
    { name: "Home", href: "#home" },
    { name: "Journey", href: "#timeline" },
    ...weddingData.events.map((e) => ({ name: e.name.split(" ")[0], href: `#event-${e.id}` })),
    { name: "RSVP", href: "#rsvp" },
  ];

  return (
    <nav className={`fixed top-0 w-full z-40 transition-all duration-300 ${scrolled ? "glassmorphism py-3" : "bg-transparent py-5"}`}>
      <div className="max-w-7xl mx-auto px-6 flex justify-between items-center">
        <a href="#home" className="font-serif text-3xl font-bold text-gold">
          {weddingData.couple.monogram}
        </a>
        
        {/* Desktop */}
        <div className="hidden md:flex gap-6 items-center">
          {links.map((link) => (
            <a key={link.name} href={link.href} className="text-sm font-medium tracking-widest uppercase hover:text-gold transition-colors">
              {link.name}
            </a>
          ))}
        </div>

        {/* Mobile Toggle */}
        <button className="md:hidden text-gold" onClick={() => setIsOpen(true)}>
          <Menu size={28} />
        </button>
      </div>

      {/* Mobile Menu */}
      <AnimatePresence>
        {isOpen && (
          <motion.div
            initial={{ x: "100%" }}
            animate={{ x: 0 }}
            exit={{ x: "100%" }}
            transition={{ type: "tween", duration: 0.4 }}
            className="fixed inset-0 bg-maroon z-50 flex flex-col items-center justify-center"
          >
            <button className="absolute top-6 right-6 text-gold" onClick={() => setIsOpen(false)}>
              <X size={32} />
            </button>
            {links.map((link, i) => (
              <motion.a
                key={link.name}
                href={link.href}
                onClick={() => setIsOpen(false)}
                initial={{ opacity: 0, y: 20 }}
                animate={{ opacity: 1, y: 0 }}
                transition={{ delay: i * 0.1 + 0.2 }}
                className="font-serif text-3xl text-gold mb-6 hover:text-white transition-colors"
              >
                {link.name}
              </motion.a>
            ))}
          </motion.div>
        )}
      </AnimatePresence>
    </nav>
  );
}


src/components/MusicPlayer.tsx

"use client";
import { useState, useRef, useEffect } from "react";
import { Music, Pause } from "lucide-react";

export default function MusicPlayer() {
  const [isPlaying, setIsPlaying] = useState(false);
  const audioRef = useRef<HTMLAudioElement | null>(null);

  useEffect(() => {
    audioRef.current = new Audio("/audio/background-music.mp3");
    audioRef.current.loop = true;
  }, []);

  const togglePlay = () => {
    if (isPlaying) {
      audioRef.current?.pause();
    } else {
      audioRef.current?.play();
    }
    setIsPlaying(!isPlaying);
  };

  return (
    <button
      onClick={togglePlay}
      className={`fixed bottom-6 right-6 z-40 p-4 rounded-full shadow-2xl transition-all duration-500 ${
        isPlaying ? "bg-gold text-maroon animate-spin-slow" : "bg-maroon text-gold hover:bg-opacity-90"
      }`}
      style={{ animationDuration: '4s' }}
    >
      {isPlaying ? <Pause size={24} /> : <Music size={24} />}
    </button>
  );
}

src/components/Countdown.tsx

TypeScript
"use client";
import { useState, useEffect } from "react";
import { motion } from "framer-motion";

export default function Countdown({ targetDate }: { targetDate: string }) {
  const [timeLeft, setTimeLeft] = useState({ days: 0, hours: 0, minutes: 0, seconds: 0 });

  useEffect(() => {
    const timer = setInterval(() => {
      const difference = new Date(targetDate).getTime() - new Date().getTime();
      if (difference > 0) {
        setTimeLeft({
          days: Math.floor(difference / (1000 * 60 * 60 * 24)),
          hours: Math.floor((difference / (1000 * 60 * 60)) % 24),
          minutes: Math.floor((difference / 1000 / 60) % 60),
          seconds: Math.floor((difference / 1000) % 60),
        });
      } else {
        clearInterval(timer);
      }
    }, 1000);
    return () => clearInterval(timer);
  }, [targetDate]);

  return (
    <motion.div 
      initial={{ opacity: 0, scale: 0.8 }}
      animate={{ opacity: 1, scale: 1 }}
      transition={{ delay: 1.8 }}
      className="flex gap-4 md:gap-8"
    >
      {Object.entries(timeLeft).map(([unit, value]) => (
        <div key={unit} className="flex flex-col items-center">
          <div className="w-16 h-16 md:w-24 md:h-24 glassmorphism border border-gold/30 rounded-lg flex items-center justify-center text-2xl md:text-4xl font-serif text-maroon shadow-lg">
            {value.toString().padStart(2, "0")}
          </div>
          <span className="mt-2 text-xs uppercase tracking-widest text-gray-500">{unit}</span>
        </div>
      ))}
    </motion.div>
  );
}

import { MapPin, Clock, Calendar } from "lucide-react";
import Image from "next/image";

interface EventProps {
  event: any;
  index: number;
}

export default function EventSection({ event, index }: EventProps) {
  const isEven = index % 2 === 0;
  
  // 3D Tilt Effect
  const x = useMotionValue(0);
  const y = useMotionValue(0);
  const rotateX = useTransform(y, [-100, 100], [15, -15]);
  const rotateY = useTransform(x, [-100, 100], [-15, 15]);

  function handleMouseMove(event: React.MouseEvent<HTMLDivElement>) {
    const rect = event.currentTarget.getBoundingClientRect();
    x.set(event.clientX - rect.left - rect.width / 2);
    y.set(event.clientY - rect.top - rect.height / 2);
  }

  function handleMouseLeave() {
    x.set(0);
    y.set(0);
  }

  return (
    <section id={`event-${event.id}`} className={`relative py-24 px-4 overflow-hidden bg-gradient-to-b ${event.color}`}>
      <div className={`max-w-6xl mx-auto flex flex-col ${isEven ? 'md:flex-row' : 'md:flex-row-reverse'} items-center gap-12 lg:gap-24`}>
        
        {/* Text Details */}

          initial={{ opacity: 0, x: isEven ? -50 : 50 }}

          transition={{ duration: 0.8 }}

Setup, Run, and Deployment Instructions
1. Setup & Run

Open your terminal and create a new Next.js app:

Bash
npx create-next-app@latest my-wedding-app --typescript --tailwind --eslint --app


Navigate into your project directory:

Bash
cd my-wedding-app


Install required packages:

Bash
npm install framer-motion lucide-react clsx tailwind-merge


Copy the code chunks above into their respective files (creating the folders inside src/ if they don't exist).

Add placeholder images (e.g. event1.jpg) into public/images/invitations/ and an MP3 file to public/audio/background-music.mp3. (The app will still work without them, but images will show a pulsing fallback background).

Start the development server:

Bash
npm run dev


Open http://localhost:3000 in your browser.

2. Vercel Deployment (Easiest Method)

Push your code to a GitHub, GitLab, or Bitbucket repository.

Go to Vercel.com and log in.

Click Add New > Project and import your repository.

Vercel will auto-detect Next.js. Leave the build settings as default.

Click Deploy. Within 2 minutes, you'll have a live URL!

3. Customization Guide

Text & Data: Open src/data/wedding.ts. Everything (Names, Dates, Venues, URLs, RSVP codes) is strictly controlled here. Update strings and the entire UI will cascade update.

Colors: Open tailwind.config.ts and src/app/globals.css. Modify the --maroon and --gold HEX codes to match your specific wedding palette.

Images: Drop your own WebP or JPG images into public/images/invitations/ ensuring they are roughly a 4:5 aspect ratio (e.g., 800x1000px) for the best 3D tilt frame result.

RSVP Backend: Inside src/components/RSVP.tsx, look for the submitRSVP function. You can replace the local state update with a fetch() call to a Google Sheets API route, Firebase, or Formspree to actually collect guest data.
